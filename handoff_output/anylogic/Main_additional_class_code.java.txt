public java.util.Random modelRng;
public int sweep;
public boolean censored;
public long runSeed;
public long startedNanos;
public int edgesAtStart;
public boolean exported;
public String stopReason;

public void initializeModel() {
    if (N < 2 || k0 < 0 || k0 >= N)
        throw new IllegalArgumentException("Need N>=2 and 0<=k0<N");
    if (!(d > 0 && d <= 1 && mu > 0 && mu <= 0.5 && phi >= 0 && phi <= 1 && h >= 0 && h <= 1))
        throw new IllegalArgumentException("Invalid d, mu, phi, or h");
    if (maxSweeps <= 0 || rep < 0 || initMode < 0 || initMode > 2)
        throw new IllegalArgumentException("Invalid cap, replication, or initialization");
    if (people.size() != N) throw new IllegalStateException("people population must have N agents");
    if (!exp.matches("[ABCD]|APPLICATION")) throw new IllegalArgumentException("exp must be A, B, C, D, or APPLICATION");
    runSeed = 1000L + rep;
    modelRng = new java.util.Random(runSeed);
    startedNanos = System.nanoTime();
    exported = false;
    stopReason = "running";
    for (int q = 0; q < N; q++) {
        Person p = people.get(q);
        p.idx = q;
        p.disconnectFromAll();
    }
sweep = 0;
censored = true;
// random graph with exactly N*k0/2 edges (same as Python reference)
int m = N * k0 / 2, added = 0;
while (added < m) {
    int a = modelRng.nextInt(N);
    int b = modelRng.nextInt(N);
    if (a == b) continue;
    Person pa = people.get(a), pb = people.get(b);
    if (pa.connectTo(pb)) added++;
}
// initial opinions
for (Person p : people) {
    if (initMode == 0) p.opinion = modelRng.nextDouble();
    else if (initMode == 1) {            // centred: median of three uniforms ~ Beta(2,2)
        double u = modelRng.nextDouble(), v = modelRng.nextDouble(), w = modelRng.nextDouble();
        p.opinion = Math.max(Math.min(u, v), Math.min(Math.max(u, v), w));
    } else {                             // bimodal around 0.25 / 0.75
        double c = (modelRng.nextDouble() < 0.5) ? 0.25 : 0.75;
        p.opinion = Math.min(1.0, Math.max(0.0, c + 0.08 * modelRng.nextGaussian()));
    }
}

    edgesAtStart = modelEdges();
}

public void modelStep() {
Person i = people.get(modelRng.nextInt(N));
int deg = i.getConnectionsNumber();
if (deg == 0) return;
Person j = (Person) i.getConnectedAgent(modelRng.nextInt(deg));
double dd = Math.abs(i.opinion - j.opinion);
if (dd < d) {
    double xi = i.opinion, xj = j.opinion;
    i.opinion = xi + mu * (xj - xi);
    j.opinion = xj + mu * (xi - xj);
} else if ((modelRng.nextDouble() < phi)) {
    if (allowIsolates || j.getConnectionsNumber() > 1) {          // forbid cuts that would isolate j
        Person k = modelPartner(i);
        if (k != null) { i.disconnectFrom(j); i.connectTo(k); }
    }
}

}

public Person modelPartner(Person i) {
boolean similar = (modelRng.nextDouble() < h);
java.util.ArrayList<Person> cand = new java.util.ArrayList<>();
for (Person k : people) {
    if (k == i || i.isConnectedTo(k)) continue;
    if (!similar || Math.abs(k.opinion - i.opinion) < d) cand.add(k);
}
if (cand.isEmpty() && similar) {                 // fall back to any non-neighbour
    for (Person k : people) if (k != i && !i.isConnectedTo(k)) cand.add(k);
}
if (cand.isEmpty()) return null;
return cand.get(modelRng.nextInt(cand.size()));

}

public boolean modelFrozen() {
double eps = 1e-3;
for (Person a : people)
    for (Person b : a.<Person>getConnections()) {
        if (a.idx >= b.idx) continue;
        double dd = Math.abs(a.opinion - b.opinion);
        if (dd < d) { if (dd > eps) return false; }
        else if (phi > 0) return false;
    }
return true;

}

public double[] modelKPIs() {
int n = N;
java.util.HashMap<Person,Integer> idx = new java.util.HashMap<>();
for (int q = 0; q < n; q++) idx.put(people.get(q), q);

// components (BFS)
int[] comp = new int[n]; java.util.Arrays.fill(comp, -1);
java.util.ArrayList<Integer> sizes = new java.util.ArrayList<>();
for (int s = 0; s < n; s++) {
    if (comp[s] != -1) continue;
    int c = sizes.size(), cnt = 0;
    java.util.ArrayDeque<Integer> dq = new java.util.ArrayDeque<>();
    dq.add(s); comp[s] = c;
    while (!dq.isEmpty()) {
        int u = dq.poll(); cnt++;
        for (Person v : people.get(u).<Person>getConnections()) {
            int vi = idx.get(v);
            if (comp[vi] == -1) { comp[vi] = c; dq.add(vi); }
        }
    }
    sizes.add(cnt);
}
int giant = 0, iso = 0;
for (int sz : sizes) { giant = Math.max(giant, sz); if (sz == 1) iso++; }

// opinion clusters: sort, split at gaps > 0.05, keep clusters with >= 5% of agents
Integer[] ord = new Integer[n];
for (int q = 0; q < n; q++) ord[q] = q;
java.util.Arrays.sort(ord, (a, b) -> Double.compare(people.get(a).opinion, people.get(b).opinion));
int[] lab = new int[n]; int cl = 0; lab[ord[0]] = 0;
for (int t = 1; t < n; t++) {
    if (people.get(ord[t]).opinion - people.get(ord[t-1]).opinion > 0.05) cl++;
    lab[ord[t]] = cl;
}
int[] csz = new int[cl + 1];
for (int q = 0; q < n; q++) csz[lab[q]]++;
int K = 0; for (int sz : csz) if (sz >= 0.05 * n) K++;

// edges, cross-cluster fraction, modularity of the opinion-cluster partition
int m = 0, cross = 0; double[] inside = new double[cl + 1], dsum = new double[cl + 1];
for (int q = 0; q < n; q++) {
    Person a = people.get(q);
    dsum[lab[q]] += a.getConnectionsNumber();
    for (Person b : a.<Person>getConnections()) {
        int bi = idx.get(b);
        if (bi > q) { m++; if (lab[bi] != lab[q]) cross++; else inside[lab[q]]++; }
    }
}
double Q = 0;
if (m > 0) for (int c2 = 0; c2 <= cl; c2++) Q += inside[c2] / m - Math.pow(dsum[c2] / (2.0 * m), 2);
double mean = 0, var = 0;
for (Person p : people) mean += p.opinion; mean /= n;
for (Person p : people) var += (p.opinion - mean) * (p.opinion - mean); var /= n;
return new double[]{ sizes.size(), giant / (double) n, iso / (double) n, K, m > 0 ? cross / (double) m : 0, Q, var };

}

public int modelEdges() {
    int count = 0;
    for (Person p : people) count += p.getConnectionsNumber();
    return count / 2;
}

public void advanceSweep() {
    if (exported) return;
    for (int q = 0; q < N; q++) modelStep();
    sweep++;
    boolean passes = sweep % 10 == 0 && modelFrozen();
    if (passes || sweep >= maxSweeps) {
        censored = !passes;
        stopReason = passes ? "numerically_frozen" : "cap";
        exportModelRun();
        finishSimulation();
    }
}

public void exportModelRun() {
    if (exported) return;
    int edgesEnd = modelEdges();
    if (edgesEnd != edgesAtStart) throw new IllegalStateException("Edge conservation failed");
    double[] m = modelKPIs();
    String header = "exp,rep,seed,N,k0,d,mu,phi,h,init,sweeps,censored,edges_start,edges_end,n_components,giant_frac,isolated_frac,K,cross_frac,Q,opinion_var,secs,cap,allow_isolates,stop_reason";
    String row = String.join(",", exp, Integer.toString(rep), Long.toString(runSeed),
        Integer.toString(N), Integer.toString(k0), Double.toString(d), Double.toString(mu),
        Double.toString(phi), Double.toString(h), Integer.toString(initMode),
        Integer.toString(sweep), censored ? "1" : "0", Integer.toString(edgesAtStart), Integer.toString(edgesEnd),
        Integer.toString((int)m[0]), Double.toString(m[1]), Double.toString(m[2]), Integer.toString((int)m[3]),
        Double.toString(m[4]), Double.toString(m[5]), Double.toString(m[6]),
        Double.toString((System.nanoTime()-startedNanos)/1.0e9), Integer.toString(maxSweeps),
        allowIsolates ? "1" : "0", stopReason);
    String id = exp+"_rep"+rep+"_n"+N+"_k"+k0+"_d"+d+"_mu"+mu+"_phi"+phi+"_h"+h+"_init"+initMode;
    java.nio.file.Path dir = java.nio.file.Paths.get(csvDirectory,
        allowIsolates ? "isolates_allowed" : "no_new_isolates", "cap"+maxSweeps);
    java.nio.file.Path target = dir.resolve(id+".csv");
    try {
        java.nio.file.Files.createDirectories(dir);
        try (java.io.BufferedWriter w = java.nio.file.Files.newBufferedWriter(target,
                java.nio.charset.StandardCharsets.UTF_8,
                java.nio.file.StandardOpenOption.CREATE, java.nio.file.StandardOpenOption.TRUNCATE_EXISTING)) {
            w.write(header); w.newLine(); w.write(row); w.newLine();
        }
    } catch (java.io.IOException e) {
        throw new RuntimeException("Cannot write CSV "+target.toAbsolutePath(), e);
    }
    exported = true;
    traceln("Run saved: "+target.toAbsolutePath());
}
